# Text extraction
pdfminer.six        # PDF text layer extraction
python-docx         # .docx / .odt reading
beautifulsoup4      # HTML parsing
lxml                # Fast HTML/XML parser (used by BeautifulSoup)

# OCR — offline (needs the tesseract and poppler system packages)
Pillow              # Image format conversion (TIFF -> JPEG, multi-page TIFF)
pytesseract         # Python wrapper for the Tesseract OCR engine
pdf2image           # PDF page -> image rasterizer (requires poppler)

# Static HTML export
markdown            # Markdown -> HTML (wiki_export.py)
